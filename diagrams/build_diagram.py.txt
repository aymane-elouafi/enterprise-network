_cache = {}
def tw(text, size, font=F_REG):
    k = (font, size)
    if k not in _cache:
        _cache[k] = ImageFont.truetype(font, size)
    return _cache[k].getlength(text)

def b64(path):
    return base64.b64encode(open(path, "rb").read()).decode()

def simple_icon(name, color, viewbox=None):
    svg = open(L + f"probe/{name}.svg").read()
    assert "<script" not in svg.lower()
    svg = svg.replace("<svg ", f'<svg fill="{color}" ', 1)
    if viewbox:
        svg = svg.replace('viewBox="0 0 24 24"', f'viewBox="{viewbox}"', 1)
    return "data:image/svg+xml;base64," + base64.b64encode(svg.encode()).decode()

def file_uri(path, mime):
    return f"data:{mime};base64," + b64(L + path)

def esc(s):
    return html.escape(s, quote=False)

out = []
def add(s):
    out.append(s)

def text(x, y, s, size=12, weight="400", fill=INK, anchor="start", mono=False, spacing=None, italic=False):
    fam = MONO if mono else SANS
    ls = f' letter-spacing="{spacing}"' if spacing else ""
    st = ' font-style="italic"' if italic else ""
    add(f'<text x="{x}" y="{y}" font-family="{fam}" font-size="{size}" font-weight="{weight}" fill="{fill}" text-anchor="{anchor}"{ls}{st}>{esc(s)}</text>')

def rect(x, y, w, h, fill="#fff", stroke=None, sw=1, rx=0, dash=None, extra=""):
    s = f' stroke="{stroke}" stroke-width="{sw}"' if stroke else ""
    d = f' stroke-dasharray="{dash}"' if dash else ""
    add(f'<rect x="{x}" y="{y}" width="{w}" height="{h}" rx="{rx}" fill="{fill}"{s}{d}{extra}/>')

def logo(key, x, y, h):
    uri, aspect, _ = LOGO[key]
    w = h * aspect
    if key == "squid":
        cid = f"clip_{int(x)}_{int(y)}"
        add(f'<clipPath id="{cid}"><circle cx="{x + h/2}" cy="{y + h/2}" r="{h/2}"/></clipPath>')
        add(f'<image x="{x}" y="{y}" width="{h}" height="{h}" href="{uri}" clip-path="url(#{cid})" preserveAspectRatio="xMidYMid slice"/>')
        add(f'<circle cx="{x + h/2}" cy="{y + h/2}" r="{h/2}" fill="none" stroke="#d9e2ec" stroke-width="1"/>')
        return h
    add(f'<image x="{x}" y="{y}" width="{w}" height="{h}" href="{uri}" preserveAspectRatio="xMidYMid meet"/>')
    return w

def chip(x, y, key, label=None, h=24, lh=15):
    """pill with logo and optional text; returns width"""
    uri, aspect, _ = LOGO[key]
    lw = lh * aspect
    pad = 8
    wtxt = tw(label, 10.5) if label else 0
    w = pad + lw + (6 + wtxt if label else 0) + pad
    if key == "dvwa":
        rect(x, y, w, h, fill="#1f2933", stroke="#1f2933", sw=1, rx=4)
    else:
        rect(x, y, w, h, fill="#f0f4f8", stroke="#d9e2ec", sw=1, rx=4)
    logo(key, x + pad, y + (h - lh) / 2, lh)
    if label:
        text(x + pad + lw + 6, y + h / 2 + 3.7, label, size=10.5, fill=INK2)
    return w

def check(x, y):

def port(cx, cy, s=10):

def polyline(points, color=LINE, sw=2.2, dash=None, arrow=False):
    pts = " ".join(f"{px},{py}" for px, py in points)
    d = f' stroke-dasharray="{dash}"' if dash else ""
    m = ' marker-end="url(#arr)"' if arrow else ""
    add(f'<polyline points="{pts}" fill="none" stroke="{color}" stroke-width="{sw}" stroke-linejoin="round" stroke-linecap="round"{d}{m}/>')

def tag(cx, cy, s, anchor="middle", size=10.5):

    text(x + w / 2, cy + 3.8, s, size=size, fill="#243b53", anchor="middle", mono=True)
